cmake_minimum_required(VERSION 3.16)
project(CampusComplaintSystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(complaint_core backend/ComplaintManager.cpp)
target_include_directories(complaint_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/backend)
add_executable(complaint_server backend/server.cpp)
target_link_libraries(complaint_server PRIVATE complaint_core)
add_executable(complaint_cli backend/cli_main.cpp)
target_link_libraries(complaint_cli PRIVATE complaint_core)
if(WIN32)
  target_link_libraries(complaint_server PRIVATE ws2_32)
endif()
